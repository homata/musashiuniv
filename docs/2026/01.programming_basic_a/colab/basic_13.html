# - もし割引率が0未満ならエラーを防ぐ

# 2. 実装（コードを書く）
def calculate_discount(price, rate):
    # 3. テスト・デバッグ（安全装置を入れる）
    if rate < 0:
        print("エラー：割引率は0以上にしてください")
        return price

    final_price = price * (1 - rate)
    return final_price
