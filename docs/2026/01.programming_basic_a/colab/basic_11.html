# ラムダ式と高階関数(sorted)の組み合わせ

sales_data = [
    {"name": "コーヒー", "price": 400},
    {"name": "ケーキ", "price": 500},
    {"name": "紅茶", "price": 450}
]

# 値段(price)の安い順に並び替えたい！
# key= の部分に「データの x を受け取ったら、xの"price"を基準にするよ」という名無しの関数を渡します。

print("安い順:", sorted_data)
