menu = ["コーヒー", "紅茶", "ケーキ"]

# 破壊的な操作（元のリストが変化します！）
menu.append("サンドイッチ")  # append: 最後に「追加」
print("追加後:", menu)

menu.remove("紅茶")          # remove: 指定したデータを「削除」
print("削除後:", menu)

# 非破壊的な操作（元のリストは変わりません）
sub_menu = menu[0:2]         # スライスで一部を切り出す
print("スライスした結果:", sub_menu)
print("元のリストはそのまま:", menu)

追加後: ['コーヒー', '紅茶', 'ケーキ', 'サンドイッチ']
削除後: ['コーヒー', 'ケーキ', 'サンドイッチ']
スライスした結果: ['コーヒー', 'ケーキ']
元のリストはそのまま: ['コーヒー', 'ケーキ', 'サンドイッチ']

# 1日の注文履歴（辞書のリスト）
daily_orders = [
    {"item": "コーヒー", "price": 450},
    {"item": "ケーキ", "price": 500},
    {"item": "コーヒー", "price": 450}
]

# 合計金額を入れる箱（最初は0円）
total_sales = 0

# for文で1つずつ注文を取り出して、値段を足していく
for order in daily_orders:
    total_sales = total_sales + order["price"]

print("本日の合計売上は", total_sales, "円です！")

本日の合計売上は 1400 円です！
